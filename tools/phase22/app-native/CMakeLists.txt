cmake_minimum_required(VERSION 3.5.0)
project(phase22)
set(CMAKE_CXX_STANDARD 17)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)
set(LIBDIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/arm64-v8a)
add_library(phase22 SHARED napi.cpp integration.c preflight.c play_policy.c play_factory.cpp)
foreach(NAME mpv avcodec.so.62 avformat.so.62 avutil.so.60 avfilter.so.11 swresample.so.6 swscale.so.9)
 if(NAME STREQUAL "mpv")
  set(FILE libmpv.so)
 else()
  set(FILE lib${NAME})
 endif()
 target_link_libraries(phase22 PUBLIC ${LIBDIR}/${FILE})
endforeach()
target_link_libraries(phase22 PUBLIC libace_napi.z.so libhilog_ndk.z.so)
target_link_options(phase22 PRIVATE -Wl,--no-undefined)
